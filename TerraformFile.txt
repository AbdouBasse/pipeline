parameters {
  booleanParam(name: 'APPLY_INFRA', defaultValue: false, description: 'Appliquer terraform apply ?')
}

pipeline {
  agent any

  environment {
    PATH = "/usr/local/bin:$PATH"
    AWS_DEFAULT_REGION = 'us-west-2'
    LOG_DIR = "/var/log/jenkins"
    BUILD_TAG = "terraform-${env.BUILD_NUMBER}"
  }

  stages {
    stage('Checkout') {
      steps {
        checkout scmGit(
          branches: [[name: '*/main']],
          userRemoteConfigs: [[
            credentialsId: 'dockerhub-credentials',
            url: 'https://github.com/AbdouBasse/pipeline.git'
          ]]
        )
      }
    }

    stage('Terraform Plan') {
      steps {
        withCredentials([
          usernamePassword(
            credentialsId: 'aws-credentials',
            usernameVariable: 'AWS_ACCESS_KEY_ID',
            passwordVariable: 'AWS_SECRET_ACCESS_KEY'
          ),
          string(credentialsId: 'aws-session-token', variable: 'AWS_SESSION_TOKEN')
        ]) {
          sh '''
            mkdir -p $LOG_DIR
            cd infrastructure
            terraform init | tee $LOG_DIR/terraform-init-${BUILD_TAG}.log
            terraform plan -var-file=terraform.tfvars | tee $LOG_DIR/terraform-plan-${BUILD_TAG}.log
          '''
        }
      }
    }

    stage('Terraform Apply') {
      when {
        expression { return params.APPLY_INFRA }
      }
      steps {
        withCredentials([
          usernamePassword(
            credentialsId: 'aws-credentials',
            usernameVariable: 'AWS_ACCESS_KEY_ID',
            passwordVariable: 'AWS_SECRET_ACCESS_KEY'
          ),
          string(credentialsId: 'aws-session-token', variable: 'AWS_SESSION_TOKEN')
        ]) {
          sh '''
            mkdir -p $LOG_DIR
            cd infrastructure
            terraform apply -auto-approve -var-file=terraform.tfvars | tee $LOG_DIR/terraform-apply-${BUILD_TAG}.log
          '''
        }
      }
    }

    stage('Promtail Log Info') {
      steps {
        echo "Logs Terraform enregistrés dans : $LOG_DIR"
        echo "Tu peux les explorer dans Grafana local avec LogQL :"
        echo "{job=\"jenkins\"} |= \"${BUILD_TAG}\""
      }
    }
  }
}
